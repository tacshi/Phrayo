cmake_minimum_required(VERSION 3.28)
project(phrayo_voice LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

get_filename_component(PHRAYO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(DEPS "${PHRAYO_ROOT}/build/android-native")
set(COSY_SOURCE "${DEPS}/cosyvoice.cpp-1616b12215407193fafb08111e7fd2939f977789")
if(NOT EXISTS "${COSY_SOURCE}/vendor/pcre2/CMakeLists.txt")
  message(FATAL_ERROR "Run python3 scripts/prepare-android-voice.py first")
endif()
set(PHRAYO_CORE_ONLY ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(COSYVOICE_NO_FRONTEND ON CACHE BOOL "" FORCE)
set(COSYVOICE_NO_AUDIO ON CACHE BOOL "" FORCE)
set(COSYVOICE_NO_ICU ON CACHE BOOL "" FORCE)
set(GGML_SOURCE_DIR "${DEPS}/ggml-e91ded11bdcd78c42f9c8d3978ff6686eb4c1226" CACHE PATH "" FORCE)
set(SIMDE_INCLUDE_DIR "${DEPS}/simde-71fd833d9666141edcd1d3c109a80e228303d8d7" CACHE PATH "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
option(PHRAYO_VULKAN "Build the Vulkan comparison backend" OFF)
set(GGML_VULKAN ${PHRAYO_VULKAN} CACHE BOOL "" FORCE)
if(ANDROID)
  set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16" CACHE STRING "Snapdragon flagship baseline" FORCE)
endif()
if(PHRAYO_VULKAN)
  set(Vulkan_INCLUDE_DIR "${DEPS}/Vulkan-Headers-ee2ec5fd83dafce291024683b50dc89219333076/include" CACHE PATH "" FORCE)
  set(SPIRV-Headers_DIR "${DEPS}/spirv-install/share/cmake/SPIRV-Headers" CACHE PATH "" FORCE)
endif()
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${COSY_SOURCE}" "${CMAKE_BINARY_DIR}/cosyvoice")
target_include_directories(cosyvoice PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
option(PHRAYO_PROFILE "Collect native stage timings" OFF)
if(PHRAYO_PROFILE)
  target_compile_definitions(cosyvoice PRIVATE PHRAYO_PROFILE=1)
endif()
if(TARGET ggml-vulkan)
  target_include_directories(ggml-vulkan PRIVATE "${DEPS}/spirv-install/include")
endif()

if(ANDROID)
  set(MECAB_SOURCE "${DEPS}/mecab-aa318f82798d72e5cb52ba5f6b10f0b780b30d05/mecab/src")
  file(GLOB MECAB_FILES "${MECAB_SOURCE}/*.cpp")
  list(FILTER MECAB_FILES EXCLUDE REGEX "/mecab(-[a-z-]+)?\\.cpp$")
  add_library(phrayo_mecab STATIC ${MECAB_FILES})
  set_target_properties(phrayo_mecab PROPERTIES CXX_STANDARD 14)
  target_include_directories(phrayo_mecab PUBLIC "${MECAB_SOURCE}")
  target_compile_options(phrayo_mecab PRIVATE -Wno-deprecated-declarations -Wno-register)
  target_compile_definitions(phrayo_mecab PRIVATE
    VERSION="0.996" PACKAGE="mecab" DIC_VERSION=102 MECAB_DEFAULT_RC="/dev/null"
    MECAB_USE_UTF8_ONLY=1 HAVE_ICONV=1 ICONV_CONST= HAVE_GETENV=1 HAVE_MMAP=1
    HAVE_SYS_MMAN_H=1 HAVE_SYS_TYPES_H=1 HAVE_SYS_STAT_H=1 HAVE_FCNTL_H=1
    HAVE_UNISTD_H=1 HAVE_PTHREAD_H=1 HAVE_LIBPTHREAD=1 HAVE_GETPAGESIZE=1
    HAVE_DIRENT_H=1 HAVE_OPENDIR=1 HAVE_GCC_ATOMIC_OPS=1 HAVE_STDINT_H=1
    SIZEOF_CHAR=1 SIZEOF_INT=4 SIZEOF_LONG=8 SIZEOF_LONG_LONG=8 SIZEOF_SHORT=2 SIZEOF_SIZE_T=8)
  add_library(phrayo_cosyvoice SHARED voice_jni.cpp)
  target_link_libraries(phrayo_cosyvoice PRIVATE cosyvoice ggml phrayo_mecab android log)
  target_link_options(phrayo_cosyvoice PRIVATE "-Wl,-z,max-page-size=16384")
else()
  add_executable(verify_voice_profile verify_voice_profile.cpp)
  target_link_libraries(verify_voice_profile PRIVATE cosyvoice)
  add_executable(benchmark_voice benchmark_voice.cpp)
  target_link_libraries(benchmark_voice PRIVATE cosyvoice ggml)
  add_subdirectory("${COSY_SOURCE}/tools/common" "${CMAKE_BINARY_DIR}/tools-common")
  add_subdirectory("${COSY_SOURCE}/tools/quantize" "${CMAKE_BINARY_DIR}/quantize")
  target_include_directories(quantize PRIVATE "${COSY_SOURCE}/vendor")
endif()
